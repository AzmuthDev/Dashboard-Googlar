={{
  (() => {
    const termoRaw = String($json['Termo de pesquisa'] || 'N/A').trim();
    const kwRaw = String($json['Palavra-chave'] || 'N/A').trim();
    const campRaw = String($json['Campanha'] || '__CLIENTE__').trim();
    const obsRaw = String($json['Observacao'] || 'Avaliar pertinência').replace(/\n/g, ' ').trim();

    const termo = termoRaw.length > 50 ? termoRaw.substring(0, 47) + '...' : termoRaw;
    const kw = kwRaw.length > 35 ? kwRaw.substring(0, 32) + '...' : kwRaw;

    const topo = `🏢 *Cliente:* __CLIENTE__\n🔍 *Termo:* "${termo}"\n🔑 *KW:* ${kw}\n💬 *Obs:* `;
    const espacoLivre = 250 - topo.length;

    let obsTratada = obsRaw;
    let campFinal = '';

    if (obsRaw.length <= espacoLivre - 25 && campRaw) {
      const sobra = espacoLivre - obsRaw.length - 5;
      if (sobra >= 10) {
        campFinal = `\n📁 ${campRaw.substring(0, sobra)}`;
      }
    } else if (obsRaw.length > espacoLivre) {
      obsTratada = obsRaw.substring(0, espacoLivre - 3) + '...';
    }

    return JSON.stringify({
      number: '__JID__',
      name: topo + obsTratada + campFinal,
      selectableCount: 1,
      values: [
        '✅ Segmentar',
        '❌ Negativar',
        '⚠️ Teste A/B',
        '🏢 Concorrente',
        '❓ Dúvida (Enviar ao Cliente)'
      ]
    });
  })()
}}
